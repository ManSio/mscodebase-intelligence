def parse_indexing_complete(line):

    """Indexing complete: 100 new/changed, 0 removed, total 3857 chunks"""

    m = re.search(r'Indexing complete.*?total\s+(\d+)\s+chunks', line)

    return int(m.group(1)) if m else None
